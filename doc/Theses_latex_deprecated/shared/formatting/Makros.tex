% New workflow: use the "acro" package.
% - Abbreviations:   \ac{...}, \acp{...} etc.
% - Symbols in text: \sym{...}  (short-only)
%
% For compatibility with old template text, \abk and the old 3-argument \sym
% are still accepted but mapped to the corresponding acro commands.

% Old: \abk{ZBA}{zweibuchstabige Abkürzung}  -> New: \ac{ZBA}
\NewDocumentCommand{\abk}{m g}{\ac{#1}}

% New: \sym{eta} -> prints the short symbol (always).
% Old: \sym{$\eta$}{eta}{Wirkungsgrad} -> mapped to \acs{eta}.
\NewDocumentCommand{\sym}{m g g}{%
  \IfNoValueTF{#2}{\acs{#1}}{\acs{#2}}%
}
