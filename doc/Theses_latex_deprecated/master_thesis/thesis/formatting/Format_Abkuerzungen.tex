% Abkürzungs- und Symbolverzeichnis (acro-based)
% -----------------------------------------------------------------------------
% This file replaces the former nomencl/makeidx workflow. Acronyms and symbols
% are declared in "Acronyms.tex" using \DeclareAcronym.

% Load acronym/symbol definitions
\input{formatting/Acronyms}

% Enable clickable links from \ac{...} to the list entry
\acsetup{make-links}

% Use acro's default first-use expansion for abbreviations.
% Symbols are printed short-only via \sym{...} (see Makros.tex).

% Custom list template to match the original layout:
% - fixed label width 2.0cm
% - compact vertical spacing
% - no page numbers
\NewAcroTemplate[list]{faps}{%
  \acroheading
  \acropreamble
  {\setlist[description]{leftmargin=!,labelwidth=2.0cm,labelsep=0.5cm,itemsep=-0.5pc,parsep=0pt,topsep=0pt}
   \begin{description}
     \renewcommand*\descriptionlabel[1]{\hspace\labelsep\normalfont ##1}
     \acronymsmapF
       {\item[\acrowrite{short}] \acrowrite{list}}
       {\item \AcroRerun}
   \end{description}}
  \acropostamble
}

% -----------------------------------------------------------------------------
% Printing macro (used by Format_Verzeichnisse.tex)
\newcommand{\FAPSPrintAcronymLists}{%
  \ifthenelse{\equal{\Verzeichnisse}{AbkuerzungSymbol}}{%
    % List of Abbreviations
    \chapter*{List of Abbreviations}
    \markboth{LIST OF ABBREVIATIONS}{LIST OF ABBREVIATIONS}
    \ifthenelse{\equal{\preChapterToC}{ja}}{\addcontentsline{toc}{chapter}{List of Abbreviations}}{}
    \printacronyms[include=abbrev,heading=none,template=faps,sort=true,display=all]

    % List of Symbols (starts on next page, like the original template)
    \newpage
    \chapter*{List of Symbols}
    \markboth{LIST OF SYMBOLS}{LIST OF SYMBOLS}
    \vspace{0.22cm}
    \ifthenelse{\equal{\preChapterToC}{ja}}{\addcontentsline{toc}{chapter}{List of Symbols}}{}
    \printacronyms[include=symbol,heading=none,template=faps,sort=true,display=all]
  }{%
    \ifthenelse{\equal{\Verzeichnisse}{Symbol}}{%
      \chapter*{List of Symbols}
      \markboth{LIST OF SYMBOLS}{LIST OF SYMBOLS}
      \ifthenelse{\equal{\preChapterToC}{ja}}{\addcontentsline{toc}{chapter}{List of Symbols}}{}
      \printacronyms[include=symbol,heading=none,template=faps,sort=true,display=all]
    }{}%
  }%
}
